cmake_minimum_required(VERSION 3.22)
project(metrys LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT EXISTS "${LLAMA_SOURCE_DIR}/include/llama.h")
 message(FATAL_ERROR "Defina LLAMA_SOURCE_DIR para o commit oficial fixado.")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(op LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_OPENSSL GGML_NATIVE GGML_OPENMP GGML_LLAMAFILE GGML_CPU_KLEIDIAI GGML_BACKEND_DL GGML_CPU_ALL_VARIANTS)
 set(${op} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(backend CUDA VULKAN OPENCL METAL BLAS RPC SYCL HIP MUSA CANN WEBGPU ZDNN)
 set(GGML_${backend} OFF CACHE BOOL "" FORCE)
endforeach()
if(ANDROID)
 if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Esta versão exige arm64-v8a.")
 endif()
 set(GGML_CPU_ARM_ARCH armv8-a CACHE STRING "" FORCE)
 add_compile_options(-march=armv8-a)
endif()
add_subdirectory("${LLAMA_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/llama" EXCLUDE_FROM_ALL)
add_library(metrys SHARED motor_local.cpp)
target_link_libraries(metrys PRIVATE llama)
target_compile_options(metrys PRIVATE -Wall -Wextra -Wpedantic)
if(ANDROID)
 target_link_options(metrys PRIVATE "-Wl,-z,max-page-size=16384")
else()
 find_package(JNI REQUIRED)
 target_include_directories(metrys PRIVATE ${JNI_INCLUDE_DIRS})
endif()
set_target_properties(metrys PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/saida")
